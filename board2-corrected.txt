var teacherCarbon = {#C#};
var teacherDouble = {#D#};

var containsCOOH = function(arr) {
  return arr.some(function(item) {
    return Array.isArray(item) &&
           item.length === 1 &&
           item[0] === 'COOH';
  });
};

var sameArray = function(a, b) {
  return JSON.stringify(a) === JSON.stringify(b);
};

var markGraph = function() {
  if (answered) {
    return;
  }

  var studentDouble = [];

  if (D.value !== '') {
    try {
      studentDouble = JSON.parse(D.value);
    } catch (e) {
      studentDouble = [];
    }
  }

  var total = 0;

  // Same as: if (carbon=C) then total:0.5
  if (Number(Carbon.value) === Number(teacherCarbon)) {
    total += 0.5;
  }

  var studentHasCOOH = containsCOOH(studentDouble);
  var teacherHasCOOH = containsCOOH(teacherDouble);

  // Same as:
  // if (member(["COOH"],dbl)=member(["COOH"],D)) then total:total+0.5
  if (studentHasCOOH === teacherHasCOOH) {
    total += 0.5;
  }

  // Same as the PRT normalization:
  // if no COOH is present, append it before comparing the full bond list to D.
  var normalizedStudent = studentDouble.map(function(item) {
    return Array.isArray(item) ? item.slice() : item;
  });

  if (!studentHasCOOH) {
    normalizedStudent.push(['COOH']);
  }

  // Same as: if (is(equal(dble,D))) then total:total+1
  if (sameArray(normalizedStudent, teacherDouble)) {
    total += 1;
  }

  answered = true;

  if (total === 2) {
    fdbk = '<span style="font-size: 1.5em; color:green;"> ✔</span> {#correct#}';
  }
  else if (total === 0) {
    fdbk = '<span style="font-size: 1.5em; color:red;"> ❌</span> {#incorrect#}';
  }
  else {
    fdbk = '<span style="font-size: 1.5em; color:red;">◗</span> {#partial#}';
  }

  board.update();
};
